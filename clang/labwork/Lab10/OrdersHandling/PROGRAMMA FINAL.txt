#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#define MAX_ORDERS 100
#define MAX_NAME_LENGTH 50
#define FILENAME_PENDING "pending_orders.bin"
#define FILENAME_CLOSED "closed_orders.bin"

typedef struct {
    char customer_name[MAX_NAME_LENGTH];
    time_t order_date;
    time_t delivery_date;
    time_t execution_date;
    int small_bottles;
    int large_bottles;
    float initial_cost;
    float discount_percentage;
    float final_cost;
    int order_status; // 0: pending, 1: ready, 2: closed
} Order;

Order pendingOrders[MAX_ORDERS];
Order readyOrders[MAX_ORDERS];
Order closedOrders[MAX_ORDERS];

int pendingOrderCount = 0;
int readyOrderCount = 0;
int closedOrderCount = 0;

const float SMALL_BOTTLE_PRICE = 0.08;
const float LARGE_BOTTLE_PRICE = 0.35;

// Function Prototypes
void addOrder();
void displayCustomerOrders();
void displayPendingOrders();
void savePendingOrders();
void loadPendingOrders();
void executeOrder();
void displayReadyOrders();
void settleOrder();
void displayClosedOrders();
void saveClosedOrders();
void loadClosedOrders();

// Utility Functions
float calculateInitialCost(int small, int large);
float calculateDiscount(float totalCost, int small, int large);
void printDate(time_t date);

int main() {
    int choice;
    
    while(1) {
        printf("\n--- Bottle Order Management System ---\n");
        printf("0. Exit Program\n");
        printf("1. Add Order\n");
        printf("2. Display Customer Orders\n");
        printf("3. Display Pending Orders\n");
        printf("4. Save Pending Orders\n");
        printf("5. Load Pending Orders\n");
        printf("6. Execute Order\n");
        printf("7. Display Ready Orders\n");
        printf("8. Settle Order\n");
        printf("9. Display Closed Orders\n");
        printf("10. Save Closed Orders\n");
        printf("11. Load Closed Orders\n");
        
        printf("Enter your choice: ");
        scanf("%d", &choice);
        
        switch(choice) {
            case 0: 
                printf("Exiting the system.\n");
                exit(0);
            case 1: addOrder(); break;
            case 2: displayCustomerOrders(); break;
            case 3: displayPendingOrders(); break;
            case 4: savePendingOrders(); break;
            case 5: loadPendingOrders(); break;
            case 6: executeOrder(); break;
            case 7: displayReadyOrders(); break;
            case 8: settleOrder(); break;
            case 9: displayClosedOrders(); break;
            case 10: saveClosedOrders(); break;
            case 11: loadClosedOrders(); break;
            default: 
                printf("Invalid choice. Please try again.\n");
        }
    }
    return 0;
}

void addOrder() {
    if (pendingOrderCount >= MAX_ORDERS) {
        printf("Maximum order limit reached.\n");
        return;
    }
    
    Order *newOrder = &pendingOrders[pendingOrderCount];
    
    printf("Enter customer name: ");
    scanf(" %[^\n]s", newOrder->customer_name);
    
    printf("Enter number of small bottles: ");
    scanf("%d", &newOrder->small_bottles);
    
    printf("Enter number of large bottles: ");
    scanf("%d", &newOrder->large_bottles);
    
    newOrder->order_date = time(NULL);
    
    printf("Enter delivery date (DD/MM/YYYY): ");
    struct tm delivery_tm = {0};
    scanf("%d/%d/%d", 
        &delivery_tm.tm_mday, 
        &delivery_tm.tm_mon, 
        &delivery_tm.tm_year);
    delivery_tm.tm_mon -= 1;
    delivery_tm.tm_year -= 1900;
    newOrder->delivery_date = mktime(&delivery_tm);
    
    newOrder->initial_cost = calculateInitialCost(
        newOrder->small_bottles, 
        newOrder->large_bottles
    );
    
    newOrder->discount_percentage = calculateDiscount(
        newOrder->initial_cost, 
        newOrder->small_bottles, 
        newOrder->large_bottles
    );
    
    newOrder->final_cost = newOrder->initial_cost * 
        (1 - newOrder->discount_percentage/100);
    
    newOrder->order_status = 0; // Pending
    
    pendingOrderCount++;
    
    printf("Order added successfully. Order Total: %.2f€\n", newOrder->final_cost);
}

void displayCustomerOrders() {
    char customerName[MAX_NAME_LENGTH];
    printf("Enter customer name: ");
    scanf(" %[^\n]s", customerName);
    
    int found = 0;
    
    for (int i = 0; i < pendingOrderCount; i++) {
        if (strcmp(pendingOrders[i].customer_name, customerName) == 0) {
            printf("\nPending Order:\n");
            printf("Small Bottles: %d\n", pendingOrders[i].small_bottles);
            printf("Large Bottles: %d\n", pendingOrders[i].large_bottles);
            printf("Initial Cost: %.2f€\n", pendingOrders[i].initial_cost);
            found = 1;
        }
    }
    
    for (int i = 0; i < readyOrderCount; i++) {
        if (strcmp(readyOrders[i].customer_name, customerName) == 0) {
            printf("\nReady Order:\n");
            printf("Small Bottles: %d\n", readyOrders[i].small_bottles);
            printf("Large Bottles: %d\n", readyOrders[i].large_bottles);
            printf("Initial Cost: %.2f€\n", readyOrders[i].initial_cost);
            found = 1;
        }
    }
    
    for (int i = 0; i < closedOrderCount; i++) {
        if (strcmp(closedOrders[i].customer_name, customerName) == 0) {
            printf("\nClosed Order:\n");
            printf("Small Bottles: %d\n", closedOrders[i].small_bottles);
            printf("Large Bottles: %d\n", closedOrders[i].large_bottles);
            printf("Final Cost: %.2f€\n", closedOrders[i].final_cost);
            found = 1;
        }
    }
    
    if (!found) {
        printf("No orders found for this customer.\n");
    }
}

void displayPendingOrders() {
    if (pendingOrderCount == 0) {
        printf("No pending orders.\n");
        return;
    }
    
    for (int i = 0; i < pendingOrderCount; i++) {
        Order *order = &pendingOrders[i];
        printf("\nPending Order %d:\n", i+1);
        printf("Customer: %s\n", order->customer_name);
        printf("Small Bottles: %d\n", order->small_bottles);
        printf("Large Bottles: %d\n", order->large_bottles);
        printf("Initial Cost: %.2f€\n", order->initial_cost);
    }
}

void savePendingOrders() {
    FILE *file = fopen(FILENAME_PENDING, "wb");
    if (file == NULL) {
        printf("Error opening file for writing.\n");
        return;
    }
    
    fwrite(&pendingOrderCount, sizeof(int), 1, file);
    fwrite(pendingOrders, sizeof(Order), pendingOrderCount, file);
    
    fclose(file);
    printf("Pending orders saved successfully.\n");
}

void loadPendingOrders() {
    FILE *file = fopen(FILENAME_PENDING, "rb");
    if (file == NULL) {
        printf("Error opening file for reading.\n");
        return;
    }
    
    fread(&pendingOrderCount, sizeof(int), 1, file);
    fread(pendingOrders, sizeof(Order), pendingOrderCount, file);
    
    fclose(file);
    printf("Pending orders loaded successfully.\n");
}

void executeOrder() {
    if (pendingOrderCount == 0) {
        printf("No pending orders to execute.\n");
        return;
    }
    
    displayPendingOrders();
    
    int orderIndex;
    printf("Enter order number to execute: ");
    scanf("%d", &orderIndex);
    
    if (orderIndex < 1 || orderIndex > pendingOrderCount) {
        printf("Invalid order number.\n");
        return;
    }
    
    // Move order to ready orders
    readyOrders[readyOrderCount] = pendingOrders[orderIndex-1];
    readyOrders[readyOrderCount].order_status = 1; // Ready
    
    // Remove from pending orders
    for (int i = orderIndex-1; i < pendingOrderCount-1; i++) {
        pendingOrders[i] = pendingOrders[i+1];
    }
    pendingOrderCount--;
    readyOrderCount++;
    
    printf("Order executed and moved to ready status.\n");
}

void displayReadyOrders() {
    if (readyOrderCount == 0) {
        printf("No orders ready for settlement.\n");
        return;
    }
    
    for (int i = 0; i < readyOrderCount; i++) {
        Order *order = &readyOrders[i];
        printf("\nReady Order %d:\n", i+1);
        printf("Customer: %s\n", order->customer_name);
        printf("Small Bottles: %d\n", order->small_bottles);
        printf("Large Bottles: %d\n", order->large_bottles);
        printf("Final Cost: %.2f€\n", order->final_cost);
    }
}

void settleOrder() {
    if (readyOrderCount == 0) {
        printf("No orders ready for settlement.\n");
        return;
    }
    
    displayReadyOrders();
    
    int orderIndex;
    printf("Enter order number to settle: ");
    scanf("%d", &orderIndex);
    
    if (orderIndex < 1 || orderIndex > readyOrderCount) {
        printf("Invalid order number.\n");
        return;
    }
    
    // Move order to closed orders
    closedOrders[closedOrderCount] = readyOrders[orderIndex-1];
    closedOrders[closedOrderCount].order_status = 2; // Closed
    closedOrders[closedOrderCount].execution_date = time(NULL);
    
    // Remove from ready orders
    for (int i = orderIndex-1; i < readyOrderCount-1; i++) {
        readyOrders[i] = readyOrders[i+1];
    }
    readyOrderCount--;
    closedOrderCount++;
    
    printf("Order settled and moved to closed status.\n");
}

void displayClosedOrders() {
    if (closedOrderCount == 0) {
        printf("No closed orders.\n");
        return;
    }
    
    for (int i = 0; i < closedOrderCount; i++) {
        Order *order = &closedOrders[i];
        printf("\nClosed Order %d:\n", i+1);
        printf("Customer: %s\n", order->customer_name);
        printf("Small Bottles: %d\n", order->small_bottles);
        printf("Large Bottles: %d\n", order->large_bottles);

    }
}

void saveClosedOrders() {
    FILE *file = fopen(FILENAME_CLOSED, "wb");
    if (file == NULL) {
        printf("Error opening file for writing.\n");

    }
    
    fwrite(&closedOrderCount, sizeof(int), 1, file);
    fwrite(closedOrders, sizeof(Order), closedOrderCount, file);

}

    FILE *file = fopen(FILENAME_CLOSED, "rb");
    if (file == NULL) {
        printf("Error opening file for reading.\n");
        return;
    }
    
    fread(&closedOrderCount, sizeof(int), 1, file);
    fread(closedOrders, sizeof(Order), closedOrderCount, file);
    
    fclose(file);

}

float calculateInitialCost(int small, int large) {
    return (small * SMALL_BOTTLE_PRICE) + (large * LARGE_BOTTLE_PRICE);
}

float calculateDiscount(float totalCost, int small, int large) {
    int totalBottles = small + large;
    
    if (totalCost > 600 || totalBottles > 1000)
        return 24.0;
    else if (totalCost > 300 || totalBottles > 1000)
        return 12.0;
    
    return 0.0;
}
